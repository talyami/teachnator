const apiKey = ""; // يتم حقن المفتاح تلقائياً

const fetchWithRetry = async (url, options, retries = 5) => {
  let delay = 1000;
  for (let i = 0; i < retries; i++) {
    try {
      const response = await fetch(url, options);
      if (!response.ok) throw new Error(`HTTP Error: ${response.status}`);
      return await response.json();
    } catch (error) {
      if (i === retries - 1) throw error;
      await new Promise(res => setTimeout(res, delay));
      delay *= 2;
    }
  }
};

// تحميل مكتبة قراءة ملفات Word ديناميكياً
const loadMammothScript = () => {
  return new Promise((resolve, reject) => {
    if (window.mammoth) return resolve();
    const script = document.createElement('script');
    script.src = 'https://cdnjs.cloudflare.com/ajax/libs/mammoth/1.4.21/mammoth.browser.min.js';
    script.onload = resolve;
    script.onerror = reject;
    document.head.appendChild(script);
  });
};

// بناء الحزمة المرسلة للذكاء الاصطناعي بناءً على نوع المادة
const buildContentsParts = (prompt, material) => {
  const parts = [{ text: prompt }];
  if (material.type === 'text') {
    parts.push({ text: `\n--- محتوى المادة ---\n${material.data}` });
  } else {
    parts.push({ inlineData: { mimeType: material.mimeType, data: material.data } });
  }
  return parts;
};

const draftQuizFromMaterial = async (material) => {
  const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.8-flash:generateContent?key=${apiKey}`;
  
  const prompt = `
    أنت معلم تفاعلي وخبير في علم الذاكرة (Mnemonics).
    قم بتحليل المادة المرفقة (سواء كانت صورة، PDF، أو نص) واستخرج منها **جميع الأسئلة بلا استثناء**.
    
    التعليمات الصارمة:
    1. حافظ على نوع السؤال تماماً: إذا كان السؤال (صح/خطأ) اجعل الخيارات ["صح", "خطأ"]. إذا كان اختيارات متعددة (MCQ)، استخرجها كما هي.
    2. استخرج "مفتاح حفظ" (Mnemonic) سحري وذكي يربط بين السؤال والإجابة بقصة من سطر واحد، تشابه صوتي، أو قاعدة مضحكة، لتثبيت المعلومة للأطفال. استخدم الإيموجي.

    يجب أن يكون الرد مصفوفة JSON فقط تحتوي على:
    - "text": نص السؤال باللغة العربية.
    - "options": مصفوفة الخيارات.
    - "correct": رقم الـ index للإجابة الصحيحة.
    - "hint": مفتاح الحفظ العبقري.
  `;

  const payload = {
    contents: [{ role: "user", parts: buildContentsParts(prompt, material) }],
    generationConfig: {
      responseMimeType: "application/json",
      thinkingConfig: {
        thinkingLevel: "high"
      },
      responseSchema: {
        type: "ARRAY",
        items: {
          type: "OBJECT",
          properties: {
            text: { type: "STRING" },
            options: { type: "ARRAY", items: { type: "STRING" } },
            correct: { type: "INTEGER" },
            hint: { type: "STRING" }
          },
          required: ["text", "options", "correct", "hint"]
        }
      }
    }
  };

  const data = await fetchWithRetry(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) });
  const parts = data.candidates?.[0]?.content?.parts || [];
  const contentPart = parts.find(p => !p.thought && p.text) || parts[parts.length - 1];
  return JSON.parse(contentPart?.text);
};

// === محرك الموسيقى التلقائي ===
let audioCtx = null;
let mainGain = null;
let bgOscillators = [];
let melodyTimer = null;
let nextNoteTime = 0;

const initAudio = () => {
  try {
    if (!audioCtx) {
      const AudioContextClass = window.AudioContext || window.webkitAudioContext;
      if (AudioContextClass) audioCtx = new AudioContextClass();
    }
    if (audioCtx && audioCtx.state === 'suspended') audioCtx.resume();
  } catch (e) {}
};

const playTone = (freq, type, duration, vol = 0.1) => {
  if (!audioCtx) return;
  try {
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = type; osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
    gain.gain.setValueAtTime(vol, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    osc.connect(gain); gain.connect(audioCtx.destination);
    osc.start(); osc.stop(audioCtx.currentTime + duration);
  } catch (e) {}
};

const playClick = () => playTone(600, 'sine', 0.1, 0.05);
const playCorrect = () => { playTone(523.25, 'sine', 0.1, 0.1); setTimeout(() => playTone(659.25, 'sine', 0.1, 0.1), 100); setTimeout(() => playTone(783.99, 'sine', 0.3, 0.1), 200); };
const playWrong = () => { playTone(300, 'triangle', 0.2, 0.1); setTimeout(() => playTone(250, 'triangle', 0.3, 0.1), 150); };
const playWin = () => { playTone(523.25, 'triangle', 0.2, 0.1); setTimeout(() => playTone(659.25, 'triangle', 0.2, 0.1), 200); setTimeout(() => playTone(783.99, 'triangle', 0.2, 0.1), 400); setTimeout(() => playTone(1046.50, 'triangle', 0.6, 0.15), 600); };

const startAmbientMusic = () => {
  initAudio();
  if (!audioCtx || mainGain) return; 
  try {
    mainGain = audioCtx.createGain(); mainGain.gain.value = 0.2; mainGain.connect(audioCtx.destination);
    const delayNode = audioCtx.createDelay(); delayNode.delayTime.value = 0.8; 
    const delayFeedback = audioCtx.createGain(); delayFeedback.gain.value = 0.4; 
    delayNode.connect(delayFeedback); delayFeedback.connect(delayNode); delayNode.connect(mainGain);
    
    const createLayer = (freq, panValue, type, vol, useDelay = false) => {
      const osc = audioCtx.createOscillator(); const gain = audioCtx.createGain();
      let panner = audioCtx.createStereoPanner ? audioCtx.createStereoPanner() : audioCtx.createGain();
      if(panner.pan) panner.pan.value = panValue;
      osc.type = type; osc.frequency.value = freq; gain.gain.value = vol;
      osc.connect(gain); gain.connect(panner); panner.connect(mainGain);
      if (useDelay) panner.connect(delayNode);
      osc.start(); bgOscillators.push(osc); return osc;
    };

    createLayer(261.63, -1, 'sine', 0.35); createLayer(269.63, 1, 'sine', 0.35);  
    createLayer(130.81, -0.5, 'triangle', 0.2); createLayer(196.00, 0.5, 'triangle', 0.15); createLayer(329.63, 0, 'sine', 0.1);        

    const scale = [261.63, 293.66, 329.63, 392.00, 440.00, 523.25, 587.33, 659.25];
    nextNoteTime = audioCtx.currentTime + 1;
    const scheduleMelody = () => {
      if (!mainGain || !audioCtx) return;
      while (nextNoteTime < audioCtx.currentTime + 0.1) {
        const freq = scale[Math.floor(Math.random() * scale.length)];
        const osc = audioCtx.createOscillator(); const gain = audioCtx.createGain();
        osc.type = 'sine'; osc.frequency.value = freq;
        gain.gain.setValueAtTime(0, nextNoteTime); gain.gain.linearRampToValueAtTime(0.2, nextNoteTime + 0.5); gain.gain.exponentialRampToValueAtTime(0.001, nextNoteTime + 5);
        osc.connect(gain); gain.connect(mainGain); gain.connect(delayNode); 
        osc.start(nextNoteTime); osc.stop(nextNoteTime + 5);
        nextNoteTime += 1.5 + Math.random() * 2.5; 
      }
      melodyTimer = setTimeout(scheduleMelody, 100);
    };
    scheduleMelody();
  } catch(e) {}
};

const stopAmbientMusic = () => {
  try {
    bgOscillators.forEach(osc => { try { osc.stop(); } catch(e){} }); bgOscillators = [];
    if (melodyTimer) clearTimeout(melodyTimer);
    if (mainGain) { mainGain.disconnect(); mainGain = null; }
  } catch(e) {}
};

// === الألعاب النارية ===
const Fireworks = ({ active }) => {
  const [particles, setParticles] = useState([]);
  useEffect(() => {
    if (active) {
      const newParticles = Array.from({ length: 30 }).map((_, i) => ({
        id: i, x: Math.random() * 100, y: Math.random() * 100, emoji: ['✨', '🎉', '🌟', '👏', '🎯', '🚀'][Math.floor(Math.random() * 6)], delay: Math.random() * 0.2,
      }));
      setParticles(newParticles);
      const timer = setTimeout(() => setParticles([]), 1500);
      return () => clearTimeout(timer);
    }
  }, [active]);
  if (!active || particles.length === 0) return null;
  return (
    <div className="absolute inset-0 pointer-events-none overflow-hidden z-50">
      {particles.map(p => (
        <div key={p.id} className="absolute animate-out fade-out slide-out-to-top-full duration-1000 ease-out" style={{ left: `${p.x}%`, top: `${p.y}%`, animationDelay: `${p.delay}s`, fontSize: `${Math.random() * 1.5 + 1}rem`, transform: `scale(${Math.random() * 0.5 + 0.8})` }}>
          {p.emoji}
        </div>
      ))}
    </div>
  );
};

export default function App() {
  const [mode, setMode] = useState('setup'); 
  const [playerName, setPlayerName] = useState('');
  const [questionsData, setQuestionsData] = useState([]);
  const [currentIndex, setCurrentIndex] = useState(0);
  const [showHint, setShowHint] = useState(false);
  const [score, setScore] = useState(0);
  const [shuffledQuestions, setShuffledQuestions] = useState([]);
  const [selectedAnswer, setSelectedAnswer] = useState(null);
  const [isAnswerChecked, setIsAnswerChecked] = useState(false);
  const [showFireworks, setShowFireworks] = useState(false);
  const [errorMsg, setErrorMsg] = useState("");
  
  const [generationStatus, setGenerationStatus] = useState('drafting'); 
  
  const [timeLeft, setTimeLeft] = useState(15 * 60);
  const [timerActive, setTimerActive] = useState(false);
  
  const [soundEnabled, setSoundEnabled] = useState(true);
  const [musicEnabled, setMusicEnabled] = useState(false);
  const fileInputRef = useRef(null);

  // --- استعادة الجلسة في حال تحديث الصفحة عند تدوير الآيباد أو الخروج المتكرر ---
  useEffect(() => {
    try {
      const savedState = sessionStorage.getItem('smartExamSession');
      if (savedState) {
        const parsed = JSON.parse(savedState);
        if (parsed.playerName) {
          setPlayerName(parsed.playerName); setMode(parsed.mode || 'setup'); setQuestionsData(parsed.questionsData || []);
          setCurrentIndex(parsed.currentIndex || 0); setScore(parsed.score || 0); setTimeLeft(parsed.timeLeft || 15 * 60);
          setTimerActive(parsed.timerActive || false); setGenerationStatus(parsed.generationStatus || 'drafting');
          if (parsed.mode === 'quiz' && parsed.shuffledQuestions) setShuffledQuestions(parsed.shuffledQuestions);
        }
      }
    } catch (e) {}
    
    // إصلاح مشكلة الـ Viewport
    let meta = document.querySelector('meta[name="viewport"]');
    if (!meta) { meta = document.createElement('meta'); meta.name = 'viewport'; document.head.appendChild(meta); }
    meta.content = 'width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no';
    
    return () => stopAmbientMusic(); 
  }, []);

  // حفظ الجلسة باستمرار
  useEffect(() => {
    try {
      if (mode !== 'generating') { // لا نحفظ الجلسة في وقت التحميل
        sessionStorage.setItem('smartExamSession', JSON.stringify({
          playerName, mode, questionsData, currentIndex, score, timeLeft, timerActive, generationStatus, shuffledQuestions
        }));
      }
    } catch (e) {}
  }, [playerName, mode, questionsData, currentIndex, score, timeLeft, timerActive, generationStatus, shuffledQuestions]);

  const handleToggleMusic = () => {
    initAudio();
    if (musicEnabled) { stopAmbientMusic(); setMusicEnabled(false); } 
    else { startAmbientMusic(); setMusicEnabled(true); }
  };

  const handleToggleSound = () => {
    initAudio();
    const newState = !soundEnabled;
    setSoundEnabled(newState);
    if (newState) setTimeout(playClick, 50); 
  };

  useEffect(() => {
    let interval = null;
    if (timerActive && timeLeft > 0) interval = setInterval(() => setTimeLeft(t => t - 1), 1000);
    else if (timeLeft === 0) setTimerActive(false);
    return () => clearInterval(interval);
  }, [timerActive, timeLeft]);

  const formatTime = (seconds) => {
    const m = Math.floor(seconds / 60).toString().padStart(2, '0');
    const s = (seconds % 60).toString().padStart(2, '0');
    return `${m}:${s}`;
  };

  const handleNameSubmit = (e) => {
    e.preventDefault();
    initAudio(); 
    if (!musicEnabled) { startAmbientMusic(); setMusicEnabled(true); }
    if (soundEnabled) playClick();
    if (playerName.trim() === '') setPlayerName('بطل');
    setMode('upload');
  };

  const resetToUpload = () => {
    if(soundEnabled) { initAudio(); playClick(); }
    setMode('upload'); setQuestionsData([]); setCurrentIndex(0); setScore(0); setSelectedAnswer(null);
    setIsAnswerChecked(false); setShowFireworks(false); setTimerActive(false); setTimeLeft(15 * 60);
    try { sessionStorage.removeItem('smartExamSession'); } catch(e){}
  };

  // معالجة جميع أنواع الملفات 
  const handleFileUpload = async (event) => {
    const file = event.target.files?.[0];
    if (!file) return;

    const isImage = file.type.startsWith('image/');
    const isPDF = file.type === 'application/pdf' || file.name.endsWith('.pdf');
    const isWord = file.type === 'application/vnd.openxmlformats-officedocument.wordprocessingml.document' || file.type === 'application/msword' || file.name.endsWith('.docx') || file.name.endsWith('.doc');

    if (!isImage && !isPDF && !isWord) {
      setErrorMsg("عذراً! يرجى رفع ملف بصيغة (صورة، PDF، أو ملف وورد Docx).");
      return;
    }

    setMode('generating');
    setGenerationStatus('drafting');
    setErrorMsg("");

    try {
      let materialPayload = null;

      if (isWord) {
        await loadMammothScript();
        const arrayBuffer = await file.arrayBuffer();
        const result = await window.mammoth.extractRawText({ arrayBuffer: arrayBuffer });
        const textContent = result.value.trim();
        
        if (!textContent) {
          throw new Error("الملف النصي فارغ. إذا كان يحتوي على صور فقط يرجى تحويله لـ PDF أو صورة.");
        }
        materialPayload = { type: 'text', data: textContent };
      } else {
        const reader = new FileReader();
        const base64Promise = new Promise(resolve => {
          reader.onloadend = () => resolve(reader.result.split(',')[1]);
        });
        reader.readAsDataURL(file);
        const base64Data = await base64Promise;
        materialPayload = { type: 'file', mimeType: file.type, data: base64Data };
      }

      const draftQuestions = await draftQuizFromMaterial(materialPayload);
      
      if (!draftQuestions || draftQuestions.length === 0) {
        throw new Error("لم يتم العثور على أسئلة واضحة.");
      }

      setGenerationStatus('revising');
      let finalQuestions = draftQuestions;
      
      try {
        const revisedQuestions = await reviseQuizDraft(materialPayload, draftQuestions);
        if (revisedQuestions && revisedQuestions.length > 0) {
          finalQuestions = revisedQuestions;
        }
      } catch (revisionError) {
        console.warn("Revision failed, using drafts.");
      }

      setQuestionsData(finalQuestions);
      setMode('intro');

    } catch (e) {
      setErrorMsg(e.message || "حدث خطأ أثناء قراءة الملف. حاول مرة أخرى بصورة أوضح.");
      setMode('upload');
    }
  };

  const handleShowHintClick = () => {
    if(soundEnabled) playClick();
    setShowHint(true);
  };

  const startLearning = () => {
    if(soundEnabled) { initAudio(); playClick(); }
    setMode('learn'); setCurrentIndex(0); setShowHint(false); setTimerActive(true);
  };

  const startQuiz = () => {
    if(soundEnabled) { initAudio(); playClick(); }
    const shuffled = [...questionsData].sort(() => Math.random() - 0.5);
    setShuffledQuestions(shuffled); setMode('quiz'); setCurrentIndex(0); setScore(0);
    setSelectedAnswer(null); setIsAnswerChecked(false); setShowFireworks(false);
    if (!timerActive) setTimerActive(true);
  };

  const handleAnswerClick = (index) => {
    if (isAnswerChecked) return; 
    initAudio(); 
    
    setSelectedAnswer(index);
    setIsAnswerChecked(true);
    
    const isCorrect = index === shuffledQuestions[currentIndex].correct;
    if (isCorrect) {
      if(soundEnabled) playCorrect();
      setScore(s => s + 1);
      setShowFireworks(true); 
    } else {
      if(soundEnabled) playWrong();
    }

    setTimeout(() => {
      setShowFireworks(false); 
      if (currentIndex < shuffledQuestions.length - 1) {
        setSelectedAnswer(null); setIsAnswerChecked(false); setCurrentIndex(c => c + 1);
      } else {
        if(soundEnabled) playWin();
        setMode('result'); setTimerActive(false);
      }
    }, 1500);
  };

  const getOptionLetter = (index) => {
    const letters = ['أ', 'ب', 'ج', 'د', 'هـ', 'و'];
    return letters[index] || (index + 1);
  };

  return (
    <div dir="rtl" className="min-h-[100dvh] bg-slate-100 flex justify-center items-center p-4 font-sans selection:bg-indigo-200">
      <div className="w-full max-w-lg bg-white rounded-3xl shadow-xl overflow-hidden relative flex flex-col min-h-[500px]">
        <Fireworks active={showFireworks} />
        
        {/* Header */}
        {mode !== 'setup' && (
          <div className="bg-gradient-to-r from-blue-600 to-indigo-600 p-4 text-white flex justify-between items-center rounded-b-3xl shadow-md z-10 relative shrink-0">
            <div className="font-bold text-lg flex items-center gap-2 truncate">
              <Sparkles className="w-5 h-5 text-yellow-300 flex-shrink-0" />
              تحدي: {playerName}
            </div>
            <div className="flex items-center gap-2 flex-shrink-0">
              {mode !== 'upload' && mode !== 'generating' && (
                <button onClick={resetToUpload} className="p-1.5 rounded-full bg-white/20 hover:bg-white/30 transition-colors" title="رفع ملف جديد">
                  <UploadCloud className="w-5 h-5" />
                </button>
              )}
              <button onClick={handleToggleMusic} className={`p-1.5 rounded-full transition-all duration-300 ${musicEnabled ? 'bg-emerald-400 text-white shadow-[0_0_15px_rgba(52,211,153,0.8)] scale-110' : 'bg-white/20 text-white/70 hover:bg-white/30'}`} title="موسيقى التركيز">
                <Music className="w-5 h-5" />
              </button>

            <div className="text-center space-y-8 animate-in fade-in zoom-in duration-500 my-auto">

                <UserCircle2 className="w-12 h-12 text-blue-600" />

                  onChange={(e) => setPlayerName(e.target.value)}
                  placeholder="اكتب اسمك هنا..."
                  className="w-full text-center text-2xl p-4 border-2 border-blue-200 rounded-2xl focus:border-blue-500 focus:ring-4 focus:ring-blue-100 outline-none transition-all"
                />
                <button 
                  type="submit"
                  className="w-full flex items-center justify-center gap-2 bg-blue-600 hover:bg-blue-700 text-white py-4 rounded-2xl font-bold text-xl transition-transform active:scale-95 shadow-lg shadow-blue-200"
                >
                  الدخول <ChevronLeft className="w-6 h-6" />
                </button>
              </form>
            </div>
          )}

          {/* UPLOAD MODE */}
          {mode === 'upload' && (
            <div className="text-center space-y-6 animate-in fade-in duration-500 my-auto">
              <div className="w-24 h-24 bg-indigo-50 text-indigo-500 rounded-full flex items-center justify-center mx-auto mb-4 border-4 border-dashed border-indigo-200 relative">
                <BrainCircuit className="w-10 h-10 absolute opacity-50 -top-2 -right-2" />
                <FileText className="w-12 h-12" />
              </div>
              <h1 className="text-2xl font-extrabold text-slate-800">مرحباً يا {playerName} 🧠</h1>
              <p className="text-slate-600 leading-relaxed text-sm">
                ارفع المادة (صورة، ملف PDF، أو ملف Word) وسأقوم بتحليلها واستخراج أسرار سحرية لحفظها!
              </p>
              
              {errorMsg && (
                <div className="p-3 bg-red-50 text-red-600 rounded-xl flex items-center gap-2 justify-center text-sm font-bold animate-in shake">
                  <AlertCircle className="w-5 h-5 flex-shrink-0" />
                  <span>{errorMsg}</span>
                </div>
              )}

              {/* التحديث هنا لقبول كافة الامتدادات بشكل واسع وشامل لأندرويد */}
              <input 
                type="file" 
                accept=".pdf,.docx,.doc,.jpg,.jpeg,.png,application/pdf,application/vnd.openxmlformats-officedocument.wordprocessingml.document,application/msword,image/*" 
                className="hidden" 
                ref={fileInputRef} 
                onChange={handleFileUpload}
              />
              <button 
                onClick={() => { initAudio(); fileInputRef.current?.click(); }}
                className="w-full flex items-center justify-center gap-3 bg-indigo-600 hover:bg-indigo-700 text-white py-5 rounded-2xl font-bold text-lg transition-transform active:scale-95 shadow-lg shadow-indigo-200"
              >
                <UploadCloud className="w-6 h-6" />
                اختيار الملف أو التقاط صورة
              </button>
            </div>
          )}

          {/* GENERATING MODE */}
          {mode === 'generating' && (
            <div className="text-center space-y-8 animate-in fade-in duration-500 my-auto">
              <div className="relative w-32 h-32 mx-auto">
                <div className="absolute inset-0 bg-blue-200 rounded-full animate-ping opacity-50"></div>
                <div className="absolute inset-0 bg-gradient-to-tr from-blue-500 to-indigo-500 rounded-full flex items-center justify-center shadow-2xl transition-all duration-500">
                  {generationStatus === 'drafting' ? (
                    <Wand2 className="w-12 h-12 text-white animate-pulse" />
                  ) : (
                    <SearchCheck className="w-12 h-12 text-white animate-bounce" />
                  )}
                </div>
              </div>
              <div className="animate-in slide-in-from-bottom-2">
                <h2 className="text-2xl font-bold text-slate-800 mb-2">
                  {generationStatus === 'drafting' ? 'أقرأ المادة الذكية... ✨' : 'المفتش الصارم يراجع... 🕵️‍♂️'}
                </h2>
                <p className="text-slate-500 font-medium animate-pulse">
                  {generationStatus === 'drafting' 
                    ? 'أبتكر مفاتيح حفظ لا تُنسى من محتوى الملف!' 
                    : 'أراجع الأسئلة، وأتأكد من دقتها 100%!'}
                </p>
              </div>
            </div>
          )}

          {/* INTRO */}
          {mode === 'intro' && (
            <div className="text-center space-y-6 animate-in fade-in zoom-in duration-500 my-auto">
              <div className="w-24 h-24 bg-emerald-100 rounded-full flex items-center justify-center mx-auto mb-4 relative">
                <Trophy className="w-12 h-12 text-emerald-600" />
              </div>
              <h1 className="text-3xl font-extrabold text-slate-800">تم تجهيز التحدي! 🚀</h1>
              <p className="text-slate-600 text-lg">
                يا بطل، حصلنا على <span className="font-bold text-indigo-600">{questionsData.length} أسئلة موثوقة</span> وجهزت لها مفاتيح حفظ سحرية. هل أنت مستعد؟
              </p>
              
              <div className="space-y-3 pt-4">
                <button 
                  onClick={startLearning}
                  className="w-full flex items-center justify-center gap-2 bg-blue-600 hover:bg-blue-700 text-white py-4 rounded-2xl font-bold text-lg transition-transform active:scale-95 shadow-lg shadow-blue-200"
                >
                  <Key className="w-6 h-6" />
                  اكتشف أسرار الحفظ (التدريب)
                </button>
                <button 
                  onClick={startQuiz}
                  className="w-full flex items-center justify-center gap-2 bg-emerald-500 hover:bg-emerald-600 text-white py-4 rounded-2xl font-bold text-lg transition-transform active:scale-95 shadow-lg shadow-emerald-200"
                >
                  <Play className="w-6 h-6" />
                  بدء الاختبار فوراً
                </button>
              </div>
            </div>
          )}

          {/* LEARN MODE */}
          {mode === 'learn' && questionsData.length > 0 && (
            <div className="flex flex-col h-full animate-in fade-in duration-300">
              <div className="flex justify-between items-center mb-6 text-slate-500 font-semibold shrink-0">
                <span>السؤال {currentIndex + 1} من {questionsData.length}</span>
                <span className="bg-blue-100 text-blue-700 px-3 py-1 rounded-full text-sm">مرحلة الفهم</span>
              </div>
              
              <div className="flex-1 flex flex-col gap-4 overflow-y-auto">
                <div className="bg-slate-50 p-6 rounded-2xl border-2 border-slate-100 text-center relative shadow-sm shrink-0">
                  <h2 className="text-xl font-bold text-slate-800 leading-relaxed mb-5">
                    {questionsData[currentIndex].text}
                  </h2>
                  <div className="space-y-3">
                    {questionsData[currentIndex].options.map((opt, i) => (
                      <div 
                        key={i} 
                        className={`p-4 rounded-xl border-2 text-right font-medium transition-colors ${
                          i === questionsData[currentIndex].correct 
                            ? 'bg-emerald-50 border-emerald-500 text-emerald-800 font-bold' 
                            : 'bg-white border-slate-200 text-slate-500 opacity-60'
                        }`}
                      >
                        {i === questionsData[currentIndex].correct && <CheckCircle2 className="inline w-6 h-6 ml-2 text-emerald-500" />}
                        <span className="font-black text-slate-400 ml-2">({getOptionLetter(i)})</span> {opt}
                      </div>
                    ))}
                  </div>
                </div>

                {!showHint ? (
                  <button 
                    onClick={handleShowHintClick}
                    className="w-full py-5 bg-gradient-to-r from-amber-200 to-orange-200 hover:from-amber-300 hover:to-orange-300 text-amber-800 font-extrabold text-lg rounded-2xl transition-all flex justify-center items-center gap-3 shadow-md hover:shadow-lg active:scale-95 shrink-0"
                  >
                    <Wand2 className="w-6 h-6" />
                    اضغط لاكتشاف السر السحري للحفظ! 🪄
                  </button>
                ) : (
                  <div className="p-6 bg-gradient-to-br from-amber-50 to-orange-50 border-2 border-amber-200 rounded-2xl flex flex-col gap-3 animate-in slide-in-from-top-2 shadow-inner shrink-0">
                    <div className="text-amber-900 font-extrabold text-lg text-center leading-relaxed whitespace-pre-wrap">
                      {questionsData[currentIndex].hint}
                    </div>
                  </div>
                )}
              </div>

              <div className="flex justify-between mt-6 pt-4 border-t border-slate-100 shrink-0">
                <button 
                  onClick={() => {
                    if(soundEnabled) playClick();
                    setCurrentIndex(prev => Math.max(0, prev - 1));
                    setShowHint(false);
                  }}
                  disabled={currentIndex === 0}
                  className="p-3 bg-slate-100 hover:bg-slate-200 text-slate-600 rounded-xl disabled:opacity-50 transition-colors"
                >
                  <ChevronRight className="w-6 h-6" />
                </button>
                
                {currentIndex === questionsData.length - 1 ? (
                  <button onClick={startQuiz} className="flex-1 mx-4 bg-emerald-500 hover:bg-emerald-600 text-white font-bold py-3 px-6 rounded-xl transition-transform active:scale-95 shadow-lg shadow-emerald-200">
                    أنا جاهز!
                  </button>
                ) : (
                  <button 
                    onClick={() => {
                      if(soundEnabled) playClick();
                      setCurrentIndex(prev => Math.min(questionsData.length - 1, prev + 1));
                      setShowHint(false);
                    }}
                    className="flex items-center gap-2 bg-blue-600 hover:bg-blue-700 text-white font-bold py-3 px-6 rounded-xl transition-colors mx-4 flex-1 justify-center"
                  >
                    السؤال التالي <ChevronLeft className="w-5 h-5" />
                  </button>
                )}
              </div>
            </div>
          )}

          {/* QUIZ MODE */}
          {mode === 'quiz' && shuffledQuestions.length > 0 && (
            <div className="flex flex-col h-full animate-in fade-in duration-300 relative z-10">
              <div className="flex justify-between items-center mb-6 shrink-0">
                <span className="text-slate-500 font-bold">سؤال {currentIndex + 1} / {shuffledQuestions.length}</span>
                <span className="bg-blue-100 text-blue-700 px-4 py-1 rounded-full font-bold">النقاط: {score}</span>
              </div>
              
              <div className="bg-white shadow-sm border-2 border-slate-100 p-6 rounded-2xl text-center mb-6 shrink-0">
                <h2 className="text-xl font-bold text-slate-800 leading-relaxed">
                  {shuffledQuestions[currentIndex].text}
                </h2>
              </div>

              <div className="space-y-3 flex-1 overflow-y-auto">
                {shuffledQuestions[currentIndex].options.map((opt, i) => {
                  let btnClass = "w-full text-right p-4 rounded-2xl border-2 font-bold text-lg transition-all duration-200 ";
                  if (!isAnswerChecked) {
                    btnClass += "bg-white border-slate-200 hover:border-blue-400 hover:bg-blue-50 text-slate-700";
                  } else {
                    if (i === shuffledQuestions[currentIndex].correct) {
                      btnClass += "bg-emerald-100 border-emerald-500 text-emerald-800 scale-[1.02] shadow-lg shadow-emerald-100";
                    } else if (i === selectedAnswer) {
                      btnClass += "bg-red-50 border-red-500 text-red-700";
                    } else {
                      btnClass += "bg-white border-slate-200 opacity-50 text-slate-400";
                    }
                  }

                  return (
                    <button 
                      key={`${currentIndex}-${i}`} 
                      onClick={() => handleAnswerClick(i)}
                      disabled={isAnswerChecked}
                      className={btnClass}
                    >
                      <span className="inline-block w-8 text-center font-black ml-2 text-slate-400">({getOptionLetter(i)})</span>
                      {opt}
                      {isAnswerChecked && i === shuffledQuestions[currentIndex].correct && <CheckCircle2 className="inline w-6 h-6 mr-2 text-emerald-500 float-left" />}
                      {isAnswerChecked && i === selectedAnswer && i !== shuffledQuestions[currentIndex].correct && <XCircle className="inline w-6 h-6 mr-2 text-red-500 float-left" />}
                    </button>
                  );
                })}
              </div>
              
              {isAnswerChecked && (
                <div className={`mt-6 p-4 rounded-xl text-center font-bold animate-in slide-in-from-bottom-2 shrink-0 ${
                  selectedAnswer === shuffledQuestions[currentIndex].correct ? 'bg-emerald-100 text-emerald-700' : 'bg-red-100 text-red-700'
                }`}>
                  {selectedAnswer === shuffledQuestions[currentIndex].correct ? 'إجابة عبقرية! 🎉' : 'لا بأس، تذكر مفتاح الحفظ لاحقاً! 💡'}
                </div>
              )}
            </div>
          )}

          {/* RESULT MODE */}
          {mode === 'result' && (
            <div className="text-center space-y-6 py-8 animate-in zoom-in duration-500 my-auto">
              <div className="relative inline-block">
                <div className="w-32 h-32 bg-blue-100 rounded-full flex items-center justify-center mx-auto mb-2">
                  {score >= (questionsData.length * 0.8) ? <Trophy className="w-16 h-16 text-yellow-500" /> : <BrainCircuit className="w-16 h-16 text-blue-500" />}
                </div>
                {score === questionsData.length && <div className="absolute top-0 right-0 animate-bounce"><Sparkles className="w-10 h-10 text-yellow-400" /></div>}
              </div>
              
              <h1 className="text-3xl font-extrabold text-slate-800">
                {score === questionsData.length ? `علامة كاملة يا بطل! 🌟` : score >= (questionsData.length * 0.8) ? `رائع جداً يا ${playerName}! 👏` : `أنت قادر على الأفضل! 💪`}
              </h1>
              <div className="text-5xl font-black text-transparent bg-clip-text bg-gradient-to-r from-blue-600 to-indigo-600">{score} / {questionsData.length}</div>
              <p className="text-slate-600 text-lg">
                {score === questionsData.length ? 'أنت مستعد 100% الآن!' : 'راجع مفاتيح الحفظ مرة أخرى وجرب التحدي.'}
              </p>

              <div className="pt-6 space-y-3">
                <button onClick={startLearning} className="w-full flex items-center justify-center gap-2 bg-blue-100 hover:bg-blue-200 text-blue-700 py-4 rounded-2xl font-bold text-lg transition-colors">
                  <Key className="w-6 h-6" /> مراجعة مفاتيح الحفظ
                </button>
                <div className="flex gap-3">
                  <button onClick={startQuiz} className="flex-1 flex items-center justify-center gap-2 bg-emerald-500 hover:bg-emerald-600 text-white py-4 rounded-2xl font-bold text-lg transition-transform active:scale-95 shadow-lg shadow-emerald-200">
                    <RotateCcw className="w-6 h-6" /> إعادة الاختبار
                  </button>
                  <button onClick={resetToUpload} className="flex items-center justify-center gap-2 bg-slate-800 hover:bg-slate-900 text-white px-6 py-4 rounded-2xl font-bold text-lg transition-transform active:scale-95 shadow-lg shadow-slate-200" title="رفع ملف جديد">
                    <UploadCloud className="w-6 h-6" />
                  </button>
                </div>
              </div>
            </div>
          )}

        </div>
      </div>
    </div>
  );
}
